   270→5. AUTHENTICATION SETUP
   271→Option A: Neon Auth (Recommended)
   272→If using Neon Auth, the users are automatically synced to neon_auth.users_sync. You just need to query that table.
   273→
   274→Create lib/auth/auth.ts:
   275→
   276→typescript
   277→import { neon } from '@neondatabase/serverless';
   278→import { db } from '@/lib/db';
   279→
   280→// Get user by email from neon_auth.users_sync
   281→export async function getUserByEmail(email: string) {
   282→  const result = await db.execute(
   283→    'SELECT * FROM neon_auth.users_sync WHERE email = $1 AND deleted_at IS NULL',
   284→    [email]
   285→  );
   286→  return result.rows[0] || null;
   287→}
   288→
   289→// Get user preferences
   290→export async function getUserPreferences(userId: string) {
   291→  const [preferences] = await db
   292→    .select()
   293→    .from(userPreferences)
   294→    .where(eq(userPreferences.userId, userId));
   295→  return preferences;
   296→}
   297→Option B: Auth.js (NextAuth v5)
   298→Create lib/auth/auth.ts:
   299→
   300→typescript
   301→import NextAuth from 'next-auth';
   302→import CredentialsProvider from 'next-auth/providers/credentials';
   303→import { db } from '@/lib/db';
   304→import { users } from '@/lib/db/schema';
   305→import { eq } from 'drizzle-orm';
   306→import { compare } from 'bcryptjs';
   307→
   308→export const { handlers, signIn, signOut, auth } = NextAuth({
   309→  providers: [
   310→    CredentialsProvider({
   311→      name: 'Credentials',
   312→      credentials: {
   313→        email: { label: 'Email', type: 'email' },
   314→        password: { label: 'Password', type: 'password' },
   315→      },
   316→      async authorize(credentials) {
   317→        if (!credentials?.email || !credentials?.password) return null;
   318→
   319→        const user = await db
   320→          .select()
   321→          .from(users)
   322→          .where(eq(users.email, credentials.email as string))
   323→          .limit(1);
   324→
   325→        if (!user[0]) return null;
   326→
   327→        const isValid = await compare(
   328→          credentials.password as string,
   329→          user[0].password
   330→        );
   331→
   332→        if (!isValid) return null;
   333→
   334→        return {
   335→          id: user[0].id,
   336→          email: user[0].email,
   337→          name: user[0].displayName,
   338→        };
   339→      },
   340→    }),
   341→  ],
   342→  pages: {
   343→    signIn: '/login',
   344→    signUp: '/signup',
   345→  },
   346→  callbacks: {
   347→    async session({ session, token }) {
   348→      if (token.sub) {
   349→        session.user.id = token.sub;
   350→      }
   351→      return session;
   352→    },
   353→    async jwt({ token, user }) {
   354→      if (user) {
   355→        token.sub = user.id;
   356→      }
   357→      return token;
   358→    },
   359→  },
   360→});

  1632→export async function POST(req: Request) {
  1633→  try {
  1634→    const { name, email, password } = await req.json();
  1635→
  1636→    // Validate input
  1637→    if (!email || !password || !name) {
  1638→      return NextResponse.json(
  1639→        { error: 'Missing required fields' },
  1640→        { status: 400 }
  1641→      );
  1642→    }
  1643→
  1644→    // Check if user exists
  1645→    const existingUser = await db
  1646→      .select()
  1647→      .from(users)
  1648→      .where(eq(users.email, email))
  1649→      .limit(1);
  1650→
  1651→    if (existingUser.length > 0) {
  1652→      return NextResponse.json(
  1653→        { error: 'User already exists' },
  1654→        { status: 409 }
  1655→      );
  1656→    }
  1657→
  1658→    // Hash password and create user
  1659→    const hashedPassword = await hash(password, 12);
  1660→    
  1661→    await db.insert(users).values({
  1662→      id: crypto.randomUUID(),
  1663→      email,
  1664→      displayName: name,
  1665→      password: hashedPassword,
  1666→    });
  1667→
  1668→    return NextResponse.json(
  1669→      { message: 'User created successfully' },
  1670→      { status: 201 }
  1671→    );
  1672→  } catch (error) {
  1673→    console.error('Signup error:', error);
  1674→    return NextResponse.json(
  1675→      { error: 'Internal server error' },
  1676→      { status: 500 }
