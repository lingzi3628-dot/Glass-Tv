    97→// lib/iptv/parser.ts
    98→
    99→import { parse } from '@iptv/playlist';
   100→
   101→export interface Channel {
   102→  id: string;
   103→  name: string;
   104→  logoUrl: string | null;
   105→  streamUrl: string;
   106→  category: string | null;
   107→  country: string | null;
   108→  language: string | null;
   109→  isVerified: boolean;
   110→  tvgId?: string;
   111→  tvgName?: string;
   112→  tvgLogo?: string;
   113→  groupTitle?: string;
   114→}
   115→
   116→export interface ParseResult {
   117→  channels: Channel[];
   118→  total: number;
   119→  source: string;
   120→}
   121→
   122→/**
   123→ * Parse an M3U/M3U8 playlist content into structured channel data
   124→ */
   125→export async function parseM3UPlaylist(content: string, sourceUrl: string): Promise<ParseResult> {
   126→  const channels: Channel[] = [];
   127→  
   128→  try {
   129→    // Parse using @iptv/playlist
   130→    const parsed = parse(content);
   131→    
   132→    for (const item of parsed.items) {
   133→      // Extract channel info from EXTINF and tags
   134→      const extinf = item.extinf || {};
   135→      
   136→      // Build channel object
   137→      const channel: Channel = {
   138→        id: generateChannelId(item.url, extinf.name || ''),
   139→        name: extinf.name || 'Unknown Channel',
   140→        logoUrl: extinf['tvg-logo'] || null,
   141→        streamUrl: item.url,
   142→        category: extinf['group-title'] || null,
   143→        country: extinf['tvg-country'] || null,
   144→        language: extinf['tvg-language'] || null,
   145→        isVerified: false,
   146→        tvgId: extinf['tvg-id'],
   147→        tvgName: extinf['tvg-name'],
   148→        tvgLogo: extinf['tvg-logo'],
   149→        groupTitle: extinf['group-title'],
   150→      };
   151→      
   152→      channels.push(channel);
   153→    }
   154→    
   155→    return {
   156→      channels,
   157→      total: channels.length,
   158→      source: sourceUrl,
   159→    };
   160→  } catch (error) {
   161→    console.error('Failed to parse M3U playlist:', error);
   162→    throw new Error(`Failed to parse playlist from ${sourceUrl}`);
   163→  }
   164→}
   165→
   166→/**
   167→ * Generate a unique channel ID
   168→ */
   169→function generateChannelId(url: string, name: string): string {
   170→  const cleanName = name.toLowerCase().replace(/[^a-z0-9]/g, '-');
   171→  const urlHash = Buffer.from(url).toString('base64').substring(0, 8);
   172→  return `${cleanName}-${urlHash}`;
   173→}
   174→
   175→/**
   176→ * Fetch and parse a playlist from a URL
   177→ */
   178→export async function fetchAndParsePlaylist(url: string): Promise<ParseResult> {
   179→  try {
   180→    const response = await fetch(url, {
   181→      headers: {
   182→        'User-Agent': 'GlassTV/1.0',
   183→      },
   184→      signal: AbortSignal.timeout(30000), // 30 second timeout
   185→    });
   186→    
   187→    if (!response.ok) {
   188→      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
   189→    }
   190→    
   191→    const content = await response.text();
   192→    return await parseM3UPlaylist(content, url);
   193→  } catch (error) {
   194→    console.error(`Failed to fetch playlist from ${url}:`, error);
   195→    throw error;
   196→  }
   197→}

   213→export async function POST(req: Request) {
   214→  try {
   215→    const { sources = DEFAULT_SOURCES } = await req.json();
   216→    
   217→    const results = {
   218→      totalChannels: 0,
   219→      newChannels: 0,
   220→      updatedChannels: 0,
   221→      failedSources: [] as string[],
   222→      sourcesProcessed: 0,
   223→    };
   224→    
   225→    // Process each source
   226→    for (const sourceUrl of sources) {
   227→      try {
   228→        const parsed = await fetchAndParsePlaylist(sourceUrl);
   229→        results.sourcesProcessed++;
   230→        
   231→        // Upsert channels to database
   232→        for (const channel of parsed.channels) {
   233→          const existing = await db
   234→            .select()
   235→            .from(channels)
   236→            .where(eq(channels.id, channel.id))
   237→            .limit(1);
   238→          
   239→          if (existing.length === 0) {
   240→            // New channel
   241→            await db.insert(channels).values({
   242→              id: channel.id,
   243→              name: channel.name,
   244→              logoUrl: channel.logoUrl,
   245→              streamUrl: channel.streamUrl,
   246→              category: channel.category,
   247→              country: channel.country,
   248→              language: channel.language,
   249→              isVerified: false,
   250→              lastUpdated: new Date(),
   251→            });
   252→            results.newChannels++;
   253→          } else {
   254→            // Update existing channel if changed
   255→            const existingChannel = existing[0];
   256→            const needsUpdate = 
   257→              existingChannel.name !== channel.name ||
   258→              existingChannel.logoUrl !== channel.logoUrl ||
   259→              existingChannel.streamUrl !== channel.streamUrl ||
   260→              existingChannel.category !== channel.category;
   261→            
   262→            if (needsUpdate) {
   263→              await db
   264→                .update(channels)
   265→                .set({
   266→                  name: channel.name,
   267→                  logoUrl: channel.logoUrl,
   268→                  streamUrl: channel.streamUrl,
   269→                  category: channel.category,
   270→                  country: channel.country,
   271→                  language: channel.language,
   272→                  lastUpdated: new Date(),
   273→                })
   274→                .where(eq(channels.id, channel.id));
   275→              results.updatedChannels++;
   276→            }
   277→          }
   278→          results.totalChannels++;
   279→        }
   280→      } catch (error) {
   281→        console.error(`Failed to process source ${sourceUrl}:`, error);
   282→        results.failedSources.push(sourceUrl);
   283→      }
   284→    }
   285→    
   286→    return NextResponse.json({
   287→      success: true,
   288→      message: 'Channel sync completed',
   289→      data: results,
   290→    });
   291→    
   292→  } catch (error) {
   293→    console.error('Channel sync error:', error);
   294→    return NextResponse.json(
   295→      { error: 'Failed to sync channels', details: String(error) },
   296→      { status: 500 }
   297→    );
   298→  }
   299→}
   300→5. CHANNELS API (GET)
   301→Create app/api/channels/route.ts:
   302→
   303→typescript
   304→// app/api/channels/route.ts
   305→
   306→import { NextResponse } from 'next/server';
   307→import { db } from '@/lib/db';
   308→import { channels } from '@/lib/db/schema';
   309→import { and, eq, ilike, desc, asc, SQL } from 'drizzle-orm';
   310→
   311→export async function GET(req: Request) {
   312→  try {
   313→    const { searchParams } = new URL(req.url);
   314→    
   315→    const page = parseInt(searchParams.get('page') || '1');
   316→    const limit = parseInt(searchParams.get('limit') || '20');
   317→    const category = searchParams.get('category');
   318→    const country = searchParams.get('country');
   319→    const search = searchParams.get('search');
   320→    const sort = searchParams.get('sort') || 'name';
   321→    const order = searchParams.get('order') || 'asc';
   322→    
   323→    const offset = (page - 1) * limit;
   324→    
   325→    // Build where clause
   326→    const conditions: SQL[] = [];
   327→    
   328→    if (category) {
   329→      conditions.push(eq(channels.category, category));
   330→    }
   331→    
   332→    if (country) {
   333→      conditions.push(eq(channels.country, country));
   334→    }
   335→    
   336→    if (search) {
   337→      conditions.push(
   338→        ilike(channels.name, `%${search}%`)
   339→      );
   340→    }
   341→    
   342→    const whereClause = conditions.length > 0 ? and(...conditions) : undefined;
   343→    
   344→    // Get total count
   345→    const countResult = await db
   346→      .select({ count: sql<number>`count(*)` })
   347→      .from(channels)
   348→      .where(whereClause);
   349→    
   350→    const total = countResult[0]?.count || 0;
   351→    
   352→    // Get channels
   353→    const sortColumn = sort === 'name' ? channels.name : channels.lastUpdated;
   354→    const sortOrder = order === 'asc' ? asc(sortColumn) : desc(sortColumn);
   355→    
   356→    const channelList = await db
   357→      .select()
   358→      .from(channels)
   359→      .where(whereClause)
   360→      .orderBy(sortOrder)
   361→      .limit(limit)
   362→      .offset(offset);
   363→    
   364→    // Get unique categories for filtering
   365→    const categories = await db
   366→      .selectDistinct({ category: channels.category })
   367→      .from(channels)
   368→      .where(sql`${channels.category} IS NOT NULL`);
   369→    
   370→    return NextResponse.json({
   371→      data: channelList,
   372→      pagination: {
   373→        page,
   374→        limit,
   375→        total,
   376→        totalPages: Math.ceil(total / limit),
   377→      },
   378→      filters: {
   379→        categories: categories.map(c => c.category).filter(Boolean),
   380→      },
   381→    });
   382→    
   383→  } catch (error) {
   384→    console.error('Channels API error:', error);
   385→    return NextResponse.json(
   386→      { error: 'Failed to fetch channels' },
   387→      { status: 500 }
   388→    );
   389→  }
   390→}
