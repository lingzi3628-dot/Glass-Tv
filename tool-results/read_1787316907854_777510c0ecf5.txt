   116→   116→export interface ParseResult {

   118→   118→  total: number;

   122→   122→/**
   123→   123→ * Parse an M3U/M3U8 playlist content into structured channel data
   124→   124→ */
   125→   125→export async function parseM3UPlaylist(content: string, sourceUrl: string): Promise<ParseResult> {
   126→   126→  const channels: Channel[] = [];
   127→   127→  
   128→   128→  try {
   129→   129→    // Parse using @iptv/playlist
   130→   130→    const parsed = parse(content);
   131→   131→    
   132→   132→    for (const item of parsed.items) {
   133→   133→      // Extract channel info from EXTINF and tags
   134→   134→      const extinf = item.extinf || {};
   135→   135→      
   136→   136→      // Build channel object
   137→   137→      const channel: Channel = {
   138→   138→        id: generateChannelId(item.url, extinf.name || ''),
   139→   139→        name: extinf.name || 'Unknown Channel',
   140→   140→        logoUrl: extinf['tvg-logo'] || null,
   141→   141→        streamUrl: item.url,
   142→   142→        category: extinf['group-title'] || null,
   143→   143→        country: extinf['tvg-country'] || null,
   144→   144→        language: extinf['tvg-language'] || null,
   145→   145→        isVerified: false,
   146→   146→        tvgId: extinf['tvg-id'],
   147→   147→        tvgName: extinf['tvg-name'],
   148→   148→        tvgLogo: extinf['tvg-logo'],
   149→   149→        groupTitle: extinf['group-title'],
   150→   150→      };
   151→   151→      
   152→   152→      channels.push(channel);
   153→   153→    }
   154→   154→    
   155→   155→    return {
   156→   156→      channels,
   157→   157→      total: channels.length,
   158→   158→      source: sourceUrl,
   159→   159→    };
   160→   160→  } catch (error) {
   161→   161→    console.error('Failed to parse M3U playlist:', error);
   162→   162→    throw new Error(`Failed to parse playlist from ${sourceUrl}`);
   163→   163→  }
   164→   164→}
   165→   165→
   166→   166→/**
   167→   167→ * Generate a unique channel ID
   168→   168→ */
   169→   169→function generateChannelId(url: string, name: string): string {
   170→   170→  const cleanName = name.toLowerCase().replace(/[^a-z0-9]/g, '-');
   171→   171→  const urlHash = Buffer.from(url).toString('base64').substring(0, 8);
   172→   172→  return `${cleanName}-${urlHash}`;
   173→   173→}
   174→   174→
   175→   175→/**
   176→   176→ * Fetch and parse a playlist from a URL
   177→   177→ */
   178→   178→export async function fetchAndParsePlaylist(url: string): Promise<ParseResult> {
   179→   179→  try {
   180→   180→    const response = await fetch(url, {
   181→   181→      headers: {
   182→   182→        'User-Agent': 'GlassTV/1.0',
   183→   183→      },
   184→   184→      signal: AbortSignal.timeout(30000), // 30 second timeout
   185→   185→    });
   186→   186→    
   187→   187→    if (!response.ok) {
   188→   188→      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
   189→   189→    }
   190→   190→    
   191→   191→    const content = await response.text();
   192→   192→    return await parseM3UPlaylist(content, url);
   193→   193→  } catch (error) {
   194→   194→    console.error(`Failed to fetch playlist from ${url}:`, error);
   195→   195→    throw error;
   196→   196→  }
   197→   197→}

   204→   204→import { NextResponse } from 'next/server';

   213→   213→export async function POST(req: Request) {
   214→   214→  try {
   215→   215→    const { sources = DEFAULT_SOURCES } = await req.json();
   216→   216→    
   217→   217→    const results = {
   218→   218→      totalChannels: 0,
   219→   219→      newChannels: 0,
   220→   220→      updatedChannels: 0,
   221→   221→      failedSources: [] as string[],
   222→   222→      sourcesProcessed: 0,
   223→   223→    };
   224→   224→    
   225→   225→    // Process each source
   226→   226→    for (const sourceUrl of sources) {
   227→   227→      try {
   228→   228→        const parsed = await fetchAndParsePlaylist(sourceUrl);
   229→   229→        results.sourcesProcessed++;
   230→   230→        
   231→   231→        // Upsert channels to database
   232→   232→        for (const channel of parsed.channels) {
   233→   233→          const existing = await db
   234→   234→            .select()
   235→   235→            .from(channels)
   236→   236→            .where(eq(channels.id, channel.id))
   237→   237→            .limit(1);
   238→   238→          
   239→   239→          if (existing.length === 0) {
   240→   240→            // New channel
   241→   241→            await db.insert(channels).values({
   242→   242→              id: channel.id,
   243→   243→              name: channel.name,
   244→   244→              logoUrl: channel.logoUrl,
   245→   245→              streamUrl: channel.streamUrl,
   246→   246→              category: channel.category,
   247→   247→              country: channel.country,
   248→   248→              language: channel.language,
   249→   249→              isVerified: false,
   250→   250→              lastUpdated: new Date(),
   251→   251→            });
   252→   252→            results.newChannels++;
   253→   253→          } else {
   254→   254→            // Update existing channel if changed
   255→   255→            const existingChannel = existing[0];
   256→   256→            const needsUpdate = 
   257→   257→              existingChannel.name !== channel.name ||
   258→   258→              existingChannel.logoUrl !== channel.logoUrl ||
   259→   259→              existingChannel.streamUrl !== channel.streamUrl ||
   260→   260→              existingChannel.category !== channel.category;
   261→   261→            
   262→   262→            if (needsUpdate) {
   263→   263→              await db
   264→   264→                .update(channels)
   265→   265→                .set({
   266→   266→                  name: channel.name,
   267→   267→                  logoUrl: channel.logoUrl,
   268→   268→                  streamUrl: channel.streamUrl,
   269→   269→                  category: channel.category,
   270→   270→                  country: channel.country,
   271→   271→                  language: channel.language,
   272→   272→                  lastUpdated: new Date(),
   273→   273→                })
   274→   274→                .where(eq(channels.id, channel.id));
   275→   275→              results.updatedChannels++;
   276→   276→            }
   277→   277→          }
   278→   278→          results.totalChannels++;
   279→   279→        }
   280→   280→      } catch (error) {
   281→   281→        console.error(`Failed to process source ${sourceUrl}:`, error);
   282→   282→        results.failedSources.push(sourceUrl);
   283→   283→      }
   284→   284→    }
   285→   285→    
   286→   286→    return NextResponse.json({
   287→   287→      success: true,
   288→   288→      message: 'Channel sync completed',
   289→   289→      data: results,
   290→   290→    });
   291→   291→    
   292→   292→  } catch (error) {
   293→   293→    console.error('Channel sync error:', error);
   294→   294→    return NextResponse.json(
   295→   295→      { error: 'Failed to sync channels', details: String(error) },
   296→   296→      { status: 500 }
   297→   297→    );
   298→   298→  }
   299→   299→}
   300→   300→5. CHANNELS API (GET)
   301→   301→Create app/api/channels/route.ts:
   302→   302→
   303→   303→typescript
   304→   304→// app/api/channels/route.ts
   305→   305→
   306→   306→import { NextResponse } from 'next/server';
   307→   307→import { db } from '@/lib/db';
   308→   308→import { channels } from '@/lib/db/schema';
   309→   309→import { and, eq, ilike, desc, asc, SQL } from 'drizzle-orm';
   310→   310→
   311→   311→export async function GET(req: Request) {
   312→   312→  try {
   313→   313→    const { searchParams } = new URL(req.url);
   314→   314→    
   315→   315→    const page = parseInt(searchParams.get('page') || '1');
   316→   316→    const limit = parseInt(searchParams.get('limit') || '20');
   317→   317→    const category = searchParams.get('category');
   318→   318→    const country = searchParams.get('country');
   319→   319→    const search = searchParams.get('search');
   320→   320→    const sort = searchParams.get('sort') || 'name';
   321→   321→    const order = searchParams.get('order') || 'asc';
   322→   322→    
   323→   323→    const offset = (page - 1) * limit;
   324→   324→    
   325→   325→    // Build where clause
   326→   326→    const conditions: SQL[] = [];
   327→   327→    
   328→   328→    if (category) {
   329→   329→      conditions.push(eq(channels.category, category));
   330→   330→    }
   331→   331→    
   332→   332→    if (country) {
   333→   333→      conditions.push(eq(channels.country, country));
   334→   334→    }
   335→   335→    
   336→   336→    if (search) {
   337→   337→      conditions.push(
   338→   338→        ilike(channels.name, `%${search}%`)
   339→   339→      );
   340→   340→    }
   341→   341→    
   342→   342→    const whereClause = conditions.length > 0 ? and(...conditions) : undefined;
   343→   343→    
   344→   344→    // Get total count
   345→   345→    const countResult = await db
   346→   346→      .select({ count: sql<number>`count(*)` })
   347→   347→      .from(channels)
   348→   348→      .where(whereClause);
   349→   349→    
   350→   350→    const total = countResult[0]?.count || 0;
   351→   351→    
   352→   352→    // Get channels
   353→   353→    const sortColumn = sort === 'name' ? channels.name : channels.lastUpdated;
   354→   354→    const sortOrder = order === 'asc' ? asc(sortColumn) : desc(sortColumn);
   355→   355→    
   356→   356→    const channelList = await db
   357→   357→      .select()
   358→   358→      .from(channels)
   359→   359→      .where(whereClause)
   360→   360→      .orderBy(sortOrder)
   361→   361→      .limit(limit)
   362→   362→      .offset(offset);
   363→   363→    
   364→   364→    // Get unique categories for filtering
   365→   365→    const categories = await db
   366→   366→      .selectDistinct({ category: channels.category })
   367→   367→      .from(channels)
   368→   368→      .where(sql`${channels.category} IS NOT NULL`);
   369→   369→    
   370→   370→    return NextResponse.json({
   371→   371→      data: channelList,
   372→   372→      pagination: {
   373→   373→        page,
   374→   374→        limit,
   375→   375→        total,
   376→   376→        totalPages: Math.ceil(total / limit),
   377→   377→      },
   378→   378→      filters: {
   379→   379→        categories: categories.map(c => c.category).filter(Boolean),
   380→   380→      },
   381→   381→    });
   382→   382→    
   383→   383→  } catch (error) {
   384→   384→    console.error('Channels API error:', error);
   385→   385→    return NextResponse.json(
   386→   386→      { error: 'Failed to fetch channels' },
   387→   387→      { status: 500 }
   388→   388→    );
   389→   389→  }
   390→   390→}
