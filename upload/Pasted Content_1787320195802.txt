// app/api/history/route.ts

import { NextResponse } from 'next/server';
import { auth } from '@/lib/auth/auth';
import { db } from '@/lib/db';
import { watchHistory, channels, users } from '@/lib/db/schema';
import { eq, desc, and, sql } from 'drizzle-orm';

export async function GET() {
  try {
    const session = await auth();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const history = await db
      .select({
        id: watchHistory.id,
        channelId: watchHistory.channelId,
        watchedAt: watchHistory.watchedAt,
        durationSeconds: watchHistory.durationSeconds,
        channel: {
          id: channels.id,
          name: channels.name,
          logoUrl: channels.logoUrl,
          category: channels.category,
        },
      })
      .from(watchHistory)
      .leftJoin(channels, eq(watchHistory.channelId, channels.id))
      .where(eq(watchHistory.userId, session.user.id))
      .orderBy(desc(watchHistory.watchedAt))
      .limit(50);

    return NextResponse.json({ data: history });
  } catch (error) {
    console.error('History API error:', error);
    return NextResponse.json(
      { error: 'Failed to fetch history' },
      { status: 500 }
    );
  }
}

export async function POST(req: Request) {
  try {
    const session = await auth();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const { channelId, durationSeconds } = await req.json();

    if (!channelId) {
      return NextResponse.json(
        { error: 'Channel ID required' },
        { status: 400 }
      );
    }

    // Insert or update watch history
    await db.insert(watchHistory).values({
      userId: session.user.id,
      channelId,
      durationSeconds: durationSeconds || 0,
      watchedAt: new Date(),
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error('History API error:', error);
    return NextResponse.json(
      { error: 'Failed to save history' },
      { status: 500 }
    );
  }
}

// hooks/useRemoteControl.ts

'use client';

import { useEffect, useRef, RefObject } from 'react';

export interface RemoteControlOptions {
  enabled?: boolean;
  onEnter?: () => void;
  onBack?: () => void;
  onArrowUp?: () => void;
  onArrowDown?: () => void;
  onArrowLeft?: () => void;
  onArrowRight?: () => void;
  onPlay?: () => void;
  onPause?: () => void;
  onVolumeUp?: () => void;
  onVolumeDown?: () => void;
  onMute?: () => void;
  onFullscreen?: () => void;
  onNumber?: (num: number) => void;
  focusSelector?: string;
  focusContainer?: RefObject<HTMLElement>;
}

export function useRemoteControl({
  enabled = true,
  onEnter,
  onBack,
  onArrowUp,
  onArrowDown,
  onArrowLeft,
  onArrowRight,
  onPlay,
  onPause,
  onVolumeUp,
  onVolumeDown,
  onMute,
  onFullscreen,
  onNumber,
  focusSelector = '[tabindex="0"], button, a, input, select, textarea',
  focusContainer,
}: RemoteControlOptions = {}) {
  const focusableElementsRef = useRef<HTMLElement[]>([]);
  const currentFocusIndex = useRef<number>(0);

  // Update focusable elements
  const updateFocusableElements = () => {
    const container = focusContainer?.current || document;
    const elements = container.querySelectorAll<HTMLElement>(focusSelector);
    focusableElementsRef.current = Array.from(elements).filter(
      (el) => !el.disabled && el.getAttribute('aria-hidden') !== 'true'
    );
  };

  // Focus next/previous element
  const focusNext = () => {
    if (focusableElementsRef.current.length === 0) return;
    currentFocusIndex.current = (currentFocusIndex.current + 1) % focusableElementsRef.current.length;
    focusableElementsRef.current[currentFocusIndex.current]?.focus();
  };

  const focusPrevious = () => {
    if (focusableElementsRef.current.length === 0) return;
    currentFocusIndex.current = (currentFocusIndex.current - 1 + focusableElementsRef.current.length) % focusableElementsRef.current.length;
    focusableElementsRef.current[currentFocusIndex.current]?.focus();
  };

  const focusFirst = () => {
    if (focusableElementsRef.current.length === 0) return;
    currentFocusIndex.current = 0;
    focusableElementsRef.current[0]?.focus();
  };

  useEffect(() => {
    if (!enabled) return;

    // Update focusable elements on mount and when DOM changes
    updateFocusableElements();
    
    const observer = new MutationObserver(() => {
      updateFocusableElements();
    });
    
    observer.observe(document.body, {
      childList: true,
      subtree: true,
    });

    // Focus first element
    setTimeout(focusFirst, 100);

    // Handle keydown events (TV remote / keyboard)
    const handleKeyDown = (e: KeyboardEvent) => {
      // Prevent default for media keys
      const mediaKeys = [' ', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Enter', 'Backspace', 'Escape', 'f', 'm'];
      if (mediaKeys.includes(e.key) || e.key.startsWith('Media')) {
        e.preventDefault();
      }

      // Number keys (0-9)
      if (e.key >= '0' && e.key <= '9') {
        onNumber?.(parseInt(e.key));
        return;
      }

      // Navigation
      switch (e.key) {
        case 'ArrowUp':
          onArrowUp?.();
          focusPrevious();
          break;
          
        case 'ArrowDown':
          onArrowDown?.();
          focusNext();
          break;
          
        case 'ArrowLeft':
          onArrowLeft?.();
          break;
          
        case 'ArrowRight':
          onArrowRight?.();
          break;
          
        case 'Enter':
          onEnter?.();
          // Simulate click on focused element
          (document.activeElement as HTMLElement)?.click?.();
          break;
          
        case 'Escape':
        case 'Backspace':
          onBack?.();
          break;
          
        case ' ':
          if (onPause && document.activeElement?.tagName !== 'INPUT') {
            onPause?.();
          }
          break;
          
        case 'MediaPlay':
          onPlay?.();
          break;
          
        case 'MediaPause':
          onPause?.();
          break;
          
        case 'MediaPlayPause':
          onPlay?.();
          break;
          
        case 'MediaStop':
          onPause?.();
          break;
          
        case 'MediaTrackPrevious':
          onVolumeDown?.();
          break;
          
        case 'MediaTrackNext':
          onVolumeUp?.();
          break;
          
        case 'f':
          onFullscreen?.();
          break;
          
        case 'm':
          onMute?.();
          break;
          
        case 'VolumeUp':
          onVolumeUp?.();
          break;
          
        case 'VolumeDown':
          onVolumeDown?.();
          break;
          
        case 'VolumeMute':
          onMute?.();
          break;
      }
    };

    // Handle focus events
    const handleFocus = (e: FocusEvent) => {
      const target = e.target as HTMLElement;
      const index = focusableElementsRef.current.indexOf(target);
      if (index !== -1) {
        currentFocusIndex.current = index;
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    document.addEventListener('focusin', handleFocus);

    return () => {
      document.removeEventListener('keydown', handleKeyDown);
      document.removeEventListener('focusin', handleFocus);
      observer.disconnect();
    };
  }, [enabled, focusSelector, focusContainer]);

  return {
    focusableElements: focusableElementsRef.current,
    currentFocusIndex: currentFocusIndex.current,
    focusFirst,
    focusNext,
    focusPrevious,
    updateFocusableElements,
  };
}

typescript
// components/player/Player.tsx

'use client';

import { useState, useEffect, useRef, useCallback } from 'react';
import Hls from 'hls.js';
import { usePlayerStore } from '@/lib/store/playerStore';
import { useRemoteControl } from '@/hooks/useRemoteControl';
import { PlayerControls } from './PlayerControls';
import { PlayerSettings } from './PlayerSettings';
import { cn } from '@/lib/utils/cn';

interface PlayerProps {
  channelId: string;
  streamUrl: string;
  channelName: string;
  channelLogo?: string | null;
  onBack?: () => void;
}

export function Player({ 
  channelId, 
  streamUrl, 
  channelName, 
  channelLogo,
  onBack,
}: PlayerProps) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);
  const hlsRef = useRef<Hls | null>(null);
  const controlsTimeoutRef = useRef<NodeJS.Timeout>();
  
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  
  const {
    isPlaying,
    isMuted,
    volume,
    currentTime,
    duration,
    buffering,
    playbackRate,
    currentLevel,
    availableLevels,
    controlsVisible,
    isFullscreen,
    isPiP,
    showSettings,
    showQualitySelector,
    setIsPlaying,
    setIsMuted,
    setVolume,
    setCurrentTime,
    setDuration,
    setBuffering,
    setPlaybackRate,
    setCurrentLevel,
    setAvailableLevels,
    setControlsVisible,
    setIsFullscreen,
    setIsPiP,
    setShowSettings,
    setShowQualitySelector,
    togglePlay,
    toggleMute,
    toggleFullscreen,
    togglePiP,
    toggleSettings,
    seekTo,
  } = usePlayerStore();

  // Initialize HLS player
  useEffect(() => {
    const video = videoRef.current;
    if (!video || !streamUrl) return;

    setLoading(true);
    setError(null);

    // Clean up existing HLS instance
    if (hlsRef.current) {
      hlsRef.current.destroy();
      hlsRef.current = null;
    }

    // Check HLS support
    if (Hls.isSupported()) {
      const hls = new Hls({
        enableWorker: true,
        lowLatencyMode: true,
        maxBufferLength: 30,
        maxMaxBufferLength: 60,
        startLevel: currentLevel,
        abrEwmaDefaultEstimate: 2000000,
        abrEwmaFastLive: 3.0,
        abrEwmaSlowLive: 9.0,
      });

      hlsRef.current = hls;
      hls.loadSource(streamUrl);
      hls.attachMedia(video);

      hls.on(Hls.Events.MANIFEST_PARSED, (_, data) => {
        setLoading(false);
        
        // Get available quality levels
        const levels = hls.levels.map((level, index) => ({
          index,
          height: level.height,
          width: level.width,
          bitrate: level.bitrate,
          label: level.height > 0 ? `${level.height}p` : `${Math.round(level.bitrate / 1000)}k`,
        }));
        
        setAvailableLevels(levels);
        
        // Try to auto-play
        video.play().catch(() => {
          // Autoplay blocked - user will need to click
        });
      });

      hls.on(Hls.Events.LEVEL_SWITCHED, (_, data) => {
        setCurrentLevel(data.level);
      });

      hls.on(Hls.Events.ERROR, (_, data) => {
        if (data.fatal) {
          setError('Failed to play stream. Please try again.');
          setLoading(false);
        }
      });

      hls.on(Hls.Events.FRAG_LOADING, () => {
        setBuffering(true);
      });

      hls.on(Hls.Events.FRAG_LOADED, () => {
        setBuffering(false);
      });

      // Set initial volume
      video.volume = volume;
      video.muted = isMuted;
    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
      // Native HLS (Safari)
      video.src = streamUrl;
      video.addEventListener('loadedmetadata', () => {
        setLoading(false);
        video.play().catch(() => {});
      });
    } else {
      setError('HLS is not supported in this browser');
      setLoading(false);
    }

    return () => {
      if (hlsRef.current) {
        hlsRef.current.destroy();
        hlsRef.current = null;
      }
    };
  }, [streamUrl]);

  // Video event listeners
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    const handlePlay = () => setIsPlaying(true);
    const handlePause = () => setIsPlaying(false);
    const handleTimeUpdate = () => setCurrentTime(video.currentTime);
    const handleDurationChange = () => setDuration(video.duration);
    const handleVolumeChange = () => {
      setIsMuted(video.muted);
      setVolume(video.volume);
    };
    const handleFullscreenChange = () => {
      setIsFullscreen(!!document.fullscreenElement);
    };

    video.addEventListener('play', handlePlay);
    video.addEventListener('pause', handlePause);
    video.addEventListener('timeupdate', handleTimeUpdate);
    video.addEventListener('durationchange', handleDurationChange);
    video.addEventListener('volumechange', handleVolumeChange);
    document.addEventListener('fullscreenchange', handleFullscreenChange);

    // Restore playback rate
    video.playbackRate = playbackRate;

    return () => {
      video.removeEventListener('play', handlePlay);
      video.removeEventListener('pause', handlePause);
      video.removeEventListener('timeupdate', handleTimeUpdate);
      video.removeEventListener('durationchange', handleDurationChange);
      video.removeEventListener('volumechange', handleVolumeChange);
      document.removeEventListener('fullscreenchange', handleFullscreenChange);
    };
  }, []);

  // Handle volume changes
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;
    video.volume = volume;
  }, [volume]);

  // Handle mute changes
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;
    video.muted = isMuted;
  }, [isMuted]);

  // Handle playback rate
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;
    video.playbackRate = playbackRate;
  }, [playbackRate]);

  // Handle quality level changes
  useEffect(() => {
    const hls = hlsRef.current;
    if (!hls) return;
    hls.currentLevel = currentLevel;
  }, [currentLevel]);

  // Save watch history on unmount
  useEffect(() => {
    return () => {
      if (currentTime > 5 && channelId) {
        fetch('/api/history', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            channelId,
            durationSeconds: Math.floor(currentTime),
          }),
        }).catch(() => {});
      }
    };
  }, [currentTime, channelId]);

  // Auto-hide controls
  useEffect(() => {
    if (controlsTimeoutRef.current) {
      clearTimeout(controlsTimeoutRef.current);
    }

    if (isPlaying && controlsVisible) {
      controlsTimeoutRef.current = setTimeout(() => {
        setControlsVisible(false);
      }, 5000);
    }

    return () => {
      if (controlsTimeoutRef.current) {
        clearTimeout(controlsTimeoutRef.current);
      }
    };
  }, [isPlaying, controlsVisible]);

  // Show controls on interaction
  const handleInteraction = () => {
    setControlsVisible(true);
    if (controlsTimeoutRef.current) {
      clearTimeout(controlsTimeoutRef.current);
    }
  };

  // Fullscreen
  const handleFullscreen = async () => {
    const container = containerRef.current;
    if (!container) return;

    try {
      if (document.fullscreenElement) {
        await document.exitFullscreen();
      } else {
        await container.requestFullscreen();
      }
      toggleFullscreen();
    } catch (error) {
      console.error('Fullscreen error:', error);
    }
  };

  // Picture-in-Picture
  const handlePiP = async () => {
    const video = videoRef.current;
    if (!video) return;

    try {
      if (document.pictureInPictureElement) {
        await document.exitPictureInPicture();
      } else {
        await video.requestPictureInPicture();
      }
      setIsPiP(!isPiP);
    } catch (error) {
      console.error('PiP error:', error);
    }
  };

  // Remote control hook
  useRemoteControl({
    enabled: true,
    onEnter: () => {
      if (showSettings) {
        toggleSettings();
      } else {
        togglePlay();
      }
    },
    onBack: () => {
      if (showSettings) {
        toggleSettings();
      } else if (onBack) {
        onBack();
      }
    },
    onArrowUp: handleInteraction,
    onArrowDown: handleInteraction,
    onArrowLeft: handleInteraction,
    onArrowRight: handleInteraction,
    onPlay: () => {
      setIsPlaying(true);
      videoRef.current?.play();
    },
    onPause: () => {
      setIsPlaying(false);
      videoRef.current?.pause();
    },
    onVolumeUp: () => {
      setVolume(Math.min(1, volume + 0.1));
    },
    onVolumeDown: () => {
      setVolume(Math.max(0, volume - 0.1));
    },
    onMute: toggleMute,
    onFullscreen: handleFullscreen,
  });

  // Keyboard shortcuts
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) {
        return;
      }

      switch (e.key) {
        case ' ':
          e.preventDefault();
          togglePlay();
          break;
        case 'f':
          handleFullscreen();
          break;
        case 'm':
          toggleMute();
          break;
        case 'ArrowUp':
          setVolume(Math.min(1, volume + 0.1));
          break;
        case 'ArrowDown':
          setVolume(Math.max(0, volume - 0.1));
          break;
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [volume]);

  return (
    <div 
      ref={containerRef}
      className="relative w-full h-screen bg-black overflow-hidden"
      onMouseMove={handleInteraction}
      onClick={handleInteraction}
    >
      {/* Video Element */}
      <video
        ref={videoRef}
        className="w-full h-full object-contain"
        playsInline
        onClick={(e) => e.stopPropagation()}
      />

      {/* Loading State */}
      {loading && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/50">
          <div className="text-center">
            <div className="w-12 h-12 border-4 border-primary-500/30 border-t-primary-500 rounded-full animate-spin mx-auto mb-4" />
            <p className="text-white/60 text-sm">Loading stream...</p>
          </div>
        </div>
      )}

      {/* Error State */}
      {error && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/80">
          <div className="text-center max-w-md px-4">
            <div className="text-4xl mb-4">⚠️</div>
            <h3 className="text-white font-semibold text-lg mb-2">Playback Error</h3>
            <p className="text-white/60 text-sm mb-4">{error}</p>
            <button
              onClick={() => window.location.reload()}
              className="px-6 py-2 bg-primary-500 text-white rounded-xl hover:bg-primary-600 transition"
            >
              Retry
            </button>
          </div>
        </div>
      )}

      {/* Buffering Overlay */}
      {buffering && !loading && !error && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/30 pointer-events-none">
          <div className="w-8 h-8 border-3 border-white/30 border-t-white rounded-full animate-spin" />
        </div>
      )}

      {/* Channel Info (Top Left) */}
      {controlsVisible && !error && (
        <div className="absolute top-4 left-4 z-20 flex items-center gap-3 glass-dark rounded-2xl px-4 py-2">
          {channelLogo ? (
            <img
              src={channelLogo}
              alt={channelName}
              className="w-8 h-8 rounded-lg object-contain bg-white/10 p-1"
            />
          ) : (
            <div className="w-8 h-8 rounded-lg bg-primary-500/20 flex items-center justify-center">
              <span className="text-sm">📺</span>
            </div>
          )}
          <div>
            <h2 className="text-white font-semibold text-sm truncate max-w-[200px]">
              {channelName}
            </h2>
            <p className="text-white/50 text-xs">Live</p>
          </div>
        </div>
      )}

      {/* Back Button (Top Left) */}
      {onBack && controlsVisible && !error && (

// app/api/channels/[channelId]/route.ts

import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { channels } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';

export async function GET(
  req: Request,
  { params }: { params: { channelId: string } }
) {
  try {
    const { channelId } = params;
    
    const channel = await db
      .select()
      .from(channels)
      .where(eq(channels.id, channelId))
      .limit(1);

    if (channel.length === 0) {
      return NextResponse.json(
        { error: 'Channel not found' },
        { status: 404 }
      );
    }

    return NextResponse.json({ data: channel[0] });
  } catch (error) {
    console.error('Channel API error:', error);
    return NextResponse.json(
      { error: 'Failed to fetch channel' },
      { status: 500 }
    );
  }
}

// components/player/KeyboardShortcuts.tsx

'use client';

import { useEffect } from 'react';

interface KeyboardShortcutsProps {
  onPlay?: () => void;
  onPause?: () => void;
  onTogglePlay?: () => void;
  onToggleMute?: () => void;
  onVolumeUp?: () => void;
  onVolumeDown?: () => void;
  onSeekForward?: (seconds: number) => void;
  onSeekBackward?: (seconds: number) => void;
  onFullscreen?: () => void;
  onPiP?: () => void;
  enabled?: boolean;
}

const SEEK_STEP = 10;

export function KeyboardShortcuts({
  onPlay,
  onPause,
  onTogglePlay,
  onToggleMute,
  onVolumeUp,
  onVolumeDown,
  onSeekForward,
  onSeekBackward,
  onFullscreen,
  onPiP,
  enabled = true,
}: KeyboardShortcutsProps) {
  useEffect(() => {
    if (!enabled) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      // Ignore if typing in input
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) {
        return;
      }

      switch (e.key) {
        case ' ':
          e.preventDefault();
          onTogglePlay?.();
          break;
          
        case 'k':
          onTogglePlay?.();
          break;
          
        case 'f':
          e.preventDefault();
          onFullscreen?.();
          break;
          
        case 'm':
          onToggleMute?.();
          break;
          
        case 'ArrowUp':
          e.preventDefault();
          onVolumeUp?.();
          break;
          
        case 'ArrowDown':
          e.preventDefault();
          onVolumeDown?.();
          break;
          
        case 'ArrowRight':
          e.preventDefault();
          onSeekForward?.(SEEK_STEP);
          break;
          
        case 'ArrowLeft':
          e.preventDefault();
          onSeekBackward?.(SEEK_STEP);
          break;
          
        case 'MediaPlay':
          onPlay?.();
          break;
          
        case 'MediaPause':
          onPause?.();
          break;
          
        case 'MediaPlayPause':
          onTogglePlay?.();
          break;
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [enabled]);

  return null;
}
