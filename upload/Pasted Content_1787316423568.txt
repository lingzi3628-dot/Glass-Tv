// lib/iptv/parser.ts

import { parse } from '@iptv/playlist';

export interface Channel {
  id: string;
  name: string;
  logoUrl: string | null;
  streamUrl: string;
  category: string | null;
  country: string | null;
  language: string | null;
  isVerified: boolean;
  tvgId?: string;
  tvgName?: string;
  tvgLogo?: string;
  groupTitle?: string;
}

export interface ParseResult {
  channels: Channel[];
  total: number;
  source: string;
}

/**
 * Parse an M3U/M3U8 playlist content into structured channel data
 */
export async function parseM3UPlaylist(content: string, sourceUrl: string): Promise<ParseResult> {
  const channels: Channel[] = [];
  
  try {
    // Parse using @iptv/playlist
    const parsed = parse(content);
    
    for (const item of parsed.items) {
      // Extract channel info from EXTINF and tags
      const extinf = item.extinf || {};
      
      // Build channel object
      const channel: Channel = {
        id: generateChannelId(item.url, extinf.name || ''),
        name: extinf.name || 'Unknown Channel',
        logoUrl: extinf['tvg-logo'] || null,
        streamUrl: item.url,
        category: extinf['group-title'] || null,
        country: extinf['tvg-country'] || null,
        language: extinf['tvg-language'] || null,
        isVerified: false,
        tvgId: extinf['tvg-id'],
        tvgName: extinf['tvg-name'],
        tvgLogo: extinf['tvg-logo'],
        groupTitle: extinf['group-title'],
      };
      
      channels.push(channel);
    }
    
    return {
      channels,
      total: channels.length,
      source: sourceUrl,
    };
  } catch (error) {
    console.error('Failed to parse M3U playlist:', error);
    throw new Error(`Failed to parse playlist from ${sourceUrl}`);
  }
}

/**
 * Generate a unique channel ID
 */
function generateChannelId(url: string, name: string): string {
  const cleanName = name.toLowerCase().replace(/[^a-z0-9]/g, '-');
  const urlHash = Buffer.from(url).toString('base64').substring(0, 8);
  return `${cleanName}-${urlHash}`;
}

/**
 * Fetch and parse a playlist from a URL
 */
export async function fetchAndParsePlaylist(url: string): Promise<ParseResult> {
  try {
    const response = await fetch(url, {
      headers: {
        'User-Agent': 'GlassTV/1.0',
      },
      signal: AbortSignal.timeout(30000), // 30 second timeout
    });
    
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    
    const content = await response.text();
    return await parseM3UPlaylist(content, url);
  } catch (error) {
    console.error(`Failed to fetch playlist from ${url}:`, error);
    throw error;
  }
}

import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { channels } from '@/lib/db/schema';
import { IPTV_SOURCES, DEFAULT_SOURCES } from '@/lib/iptv/sources';
import { fetchAndParsePlaylist, type Channel } from '@/lib/iptv/parser';
import { eq, and } from 'drizzle-orm';

export const maxDuration = 300; // 5 minutes for large playlists

export async function POST(req: Request) {
  try {
    const { sources = DEFAULT_SOURCES } = await req.json();
    
    const results = {
      totalChannels: 0,
      newChannels: 0,
      updatedChannels: 0,
      failedSources: [] as string[],
      sourcesProcessed: 0,
    };
    
    // Process each source
    for (const sourceUrl of sources) {
      try {
        const parsed = await fetchAndParsePlaylist(sourceUrl);
        results.sourcesProcessed++;
        
        // Upsert channels to database
        for (const channel of parsed.channels) {
          const existing = await db
            .select()
            .from(channels)
            .where(eq(channels.id, channel.id))
            .limit(1);
          
          if (existing.length === 0) {
            // New channel
            await db.insert(channels).values({
              id: channel.id,
              name: channel.name,
              logoUrl: channel.logoUrl,
              streamUrl: channel.streamUrl,
              category: channel.category,
              country: channel.country,
              language: channel.language,
              isVerified: false,
              lastUpdated: new Date(),
            });
            results.newChannels++;
          } else {
            // Update existing channel if changed
            const existingChannel = existing[0];
            const needsUpdate = 
              existingChannel.name !== channel.name ||
              existingChannel.logoUrl !== channel.logoUrl ||
              existingChannel.streamUrl !== channel.streamUrl ||
              existingChannel.category !== channel.category;
            
            if (needsUpdate) {
              await db
                .update(channels)
                .set({
                  name: channel.name,
                  logoUrl: channel.logoUrl,
                  streamUrl: channel.streamUrl,
                  category: channel.category,
                  country: channel.country,
                  language: channel.language,
                  lastUpdated: new Date(),
                })
                .where(eq(channels.id, channel.id));
              results.updatedChannels++;
            }
          }
          results.totalChannels++;
        }
      } catch (error) {
        console.error(`Failed to process source ${sourceUrl}:`, error);
        results.failedSources.push(sourceUrl);
      }
    }
    
    return NextResponse.json({
      success: true,
      message: 'Channel sync completed',
      data: results,
    });
    
  } catch (error) {
    console.error('Channel sync error:', error);
    return NextResponse.json(
      { error: 'Failed to sync channels', details: String(error) },
      { status: 500 }
    );
  }
}
5. CHANNELS API (GET)
Create app/api/channels/route.ts:

typescript
// app/api/channels/route.ts

import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { channels } from '@/lib/db/schema';
import { and, eq, ilike, desc, asc, SQL } from 'drizzle-orm';

export async function GET(req: Request) {
  try {
    const { searchParams } = new URL(req.url);
    
    const page = parseInt(searchParams.get('page') || '1');
    const limit = parseInt(searchParams.get('limit') || '20');
    const category = searchParams.get('category');
    const country = searchParams.get('country');
    const search = searchParams.get('search');
    const sort = searchParams.get('sort') || 'name';
    const order = searchParams.get('order') || 'asc';
    
    const offset = (page - 1) * limit;
    
    // Build where clause
    const conditions: SQL[] = [];
    
    if (category) {
      conditions.push(eq(channels.category, category));
    }
    
    if (country) {
      conditions.push(eq(channels.country, country));
    }
    
    if (search) {
      conditions.push(
        ilike(channels.name, `%${search}%`)
      );
    }
    
    const whereClause = conditions.length > 0 ? and(...conditions) : undefined;
    
    // Get total count
    const countResult = await db
      .select({ count: sql<number>`count(*)` })
      .from(channels)
      .where(whereClause);
    
    const total = countResult[0]?.count || 0;
    
    // Get channels
    const sortColumn = sort === 'name' ? channels.name : channels.lastUpdated;
    const sortOrder = order === 'asc' ? asc(sortColumn) : desc(sortColumn);
    
    const channelList = await db
      .select()
      .from(channels)
      .where(whereClause)
      .orderBy(sortOrder)
      .limit(limit)
      .offset(offset);
    
    // Get unique categories for filtering
    const categories = await db
      .selectDistinct({ category: channels.category })
      .from(channels)
      .where(sql`${channels.category} IS NOT NULL`);
    
    return NextResponse.json({
      data: channelList,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
      filters: {
        categories: categories.map(c => c.category).filter(Boolean),
      },
    });
    
  } catch (error) {
    console.error('Channels API error:', error);
    return NextResponse.json(
      { error: 'Failed to fetch channels' },
      { status: 500 }
    );
  }
}

import { useState, useEffect, useCallback } from 'react';

interface ChannelGridProps {
  initialChannels?: Channel[];
  initialCategories?: string[];
  onChannelSelect?: (channel: Channel) => void;
  showPreview?: boolean;
}

export function ChannelGrid({ 
  initialChannels = [], 
  initialCategories = [],
  onChannelSelect,
  showPreview = true,
}: ChannelGridProps) {
  const router = useRouter();
  const [channels, setChannels] = useState<Channel[]>(initialChannels);
  const [categories, setCategories] = useState<string[]>(initialCategories);
  const [loading, setLoading] = useState(false);
  const [loadingMore, setLoadingMore] = useState(false);
  const [page, setPage] = useState(1);
  const [totalPages, setTotalPages] = useState(1);
  const [search, setSearch] = useState('');
  const [debouncedSearch] = useDebounce(search, 300);
  const [selectedCategory, setSelectedCategory] = useState<string>('');
  const [showFilters, setShowFilters] = useState(false);
  const [hasMore, setHasMore] = useState(true);

  const fetchChannels = useCallback(async (reset = false) => {
    const currentPage = reset ? 1 : page;
    setLoading(reset);
    setLoadingMore(!reset);

    try {
      const params = new URLSearchParams({
        page: String(currentPage),
        limit: '20',
        ...(selectedCategory && { category: selectedCategory }),
        ...(debouncedSearch && { search: debouncedSearch }),
      });

      const response = await fetch(`/api/channels?${params}`);
      const data = await response.json();

      if (reset) {
        setChannels(data.data);
        setTotalPages(data.pagination.totalPages);
        setCategories(data.filters.categories || []);
      } else {
        setChannels(prev => [...prev, ...data.data]);
      }

      setHasMore(currentPage < data.pagination.totalPages);
      setPage(data.pagination.page);
    } catch (error) {
      console.error('Failed to fetch channels:', error);
    } finally {
      setLoading(false);
      setLoadingMore(false);
    }
  }, [page, selectedCategory, debouncedSearch]);

  // Initial load and filter changes
  useEffect(() => {
    fetchChannels(true);
  }, [selectedCategory, debouncedSearch]);

  // Load more
  const loadMore = () => {
    if (hasMore && !loadingMore) {
      setPage(prev => prev + 1);
      fetchChannels(false);
    }
  };

  const handleChannelClick = (channel: Channel) => {
    if (onChannelSelect) {
      onChannelSelect(channel);
    } else {
      router.push(`/player/${channel.id}`);
    }
  };

  // Clear filters
  const clearFilters = () => {
    setSearch('');
    setSelectedCategory('');
    setShowFilters(false);
  };

  return (
    <div className="space-y-6">
      {/* Search & Filters Bar */}
      <div className="flex flex-col sm:flex-row gap-4">
        <div className="relative flex-1">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
          <input
            type="text"
            placeholder="Search channels..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="w-full pl-10 pr-4 py-3 bg-white border border-gray-200 rounded-xl focus:outline-none focus:ring-2 focus:ring-primary-500 focus:border-transparent text-sm"
          />
          {search && (
            <button
              onClick={() => setSearch('')}
              className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
            >
              <X className="w-4 h-4" />
            </button>
          )}
        </div>

        <div className="flex gap-2">
          <GlassButton
            variant="secondary"
            size="sm"
            onClick={() => setShowFilters(!showFilters)}
            className="flex items-center gap-2"
          >
            <Filter className="w-4 h-4" />
            Filters
            <ChevronDown className={cn(
              'w-4 h-4 transition-transform',
              showFilters && 'rotate-180'
            )} />
          </GlassButton>

          {(selectedCategory || search) && (
            <GlassButton
              variant="secondary"
              size="sm"
              onClick={clearFilters}
              className="text-red-500 hover:text-red-600"
            >
              Clear
            </GlassButton>
          )}
        </div>
      </div>

      {/* Filter Chips */}
      {showFilters && categories.length > 0 && (
        <div className="flex flex-wrap gap-2 p-4 bg-white rounded-2xl border border-gray-100 shadow-sm">
          <span className="text-sm font-medium text-gray-700 mr-2">Categories:</span>
          <button
            onClick={() => setSelectedCategory('')}
            className={cn(
              'px-3 py-1.5 rounded-full text-sm transition-all',
              !selectedCategory
                ? 'bg-primary-500 text-white'
                : 'bg-gray-100 text-gray-600 hover:bg-gray-200'
            )}
          >
            All
          </button>
          {categories.map((cat) => (
            <button
              key={cat}
              onClick={() => setSelectedCategory(cat)}
              className={cn(
                'px-3 py-1.5 rounded-full text-sm transition-all',
                selectedCategory === cat
                  ? 'bg-primary-500 text-white'
                  : 'bg-gray-100 text-gray-600 hover:bg-gray-200'
              )}
            >
              {cat}
            </button>
          ))}
        </div>
      )}

      {/* Results count */}
      <div className="flex justify-between items-center text-sm text-gray-500">
        <span>{channels.length} channels</span>
        {loading && <span>Loading...</span>}
      </div>

      {/* Channel Grid */}
      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
        {channels.map((channel) => (
          <ChannelCard
            key={channel.id}
            channel={channel}
            onClick={() => handleChannelClick(channel)}
          />
        ))}
      </div>

      {/* Load More */}
      {hasMore && (
        <div className="flex justify-center pt-4">
          <GradientButton
            size="sm"
            onClick={loadMore}
            disabled={loadingMore}
          >
            {loadingMore ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin mr-2" />
                Loading...
              </>
            ) : (
              'Load More'
            )}
          </GradientButton>
        </div>
      )}

      {/* Empty state */}
      {!loading && channels.length === 0 && (
        <div className="text-center py-12">
          <div className="text-4xl mb-4">📺</div>
          <h3 className="text-lg font-medium text-gray-900">No channels found</h3>
          <p className="text-gray-500 mt-1">
            Try adjusting your search or filters
          </p>
          <GlassButton
            variant="secondary"
            className="mt-4"
            onClick={clearFilters}
          >
            Clear all filters
          </GlassButton>
        </div>
      )}
    </div>
  );
}

// components/popup/PreviewPopup.tsx

'use client';

import { useState, useEffect, useRef } from 'react';
import { useRouter } from 'next/navigation';
import { X, Play, Eye, Clock, Volume2, VolumeX, Maximize2 } from 'lucide-react';
import Hls from 'hls.js';
import { GlassCard } from '@/components/ui/GlassCard';
import { GradientButton } from '@/components/ui/GradientButton';
import { GlassButton } from '@/components/ui/GlassButton';
import { cn } from '@/lib/utils/cn';

interface Channel {
  id: string;
  name: string;
  logoUrl: string | null;
  streamUrl: string;
  category: string | null;
  country: string | null;
  language: string | null;
  isVerified: boolean;
}

interface PreviewPopupProps {
  channel: Channel;
  onWatch: (channelId: string) => void;
  onDismiss: () => void;
  autoDismissSeconds?: number;
  viewerCount?: number;
  currentShow?: string;
}

export function PreviewPopup({
  channel,
  onWatch,
  onDismiss,
  autoDismissSeconds = 10,
  viewerCount = 0,
  currentShow,
}: PreviewPopupProps) {
  const router = useRouter();
  const videoRef = useRef<HTMLVideoElement>(null);
  const hlsRef = useRef<Hls | null>(null);
  const [progress, setProgress] = useState(100);
  const [isPlaying, setIsPlaying] = useState(false);
  const [isMuted, setIsMuted] = useState(true);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const progressInterval = useRef<NodeJS.Timeout>();

  // Initialize HLS player
  useEffect(() => {
    const video = videoRef.current;
    if (!video || !channel.streamUrl) return;

    setIsLoading(true);
    setError(null);

    // Check if HLS is supported
    if (Hls.isSupported()) {
      const hls = new Hls({
        enableWorker: true,
        lowLatencyMode: true,
        maxBufferLength: 5,
        maxMaxBufferLength: 10,
        startLevel: -1, // Auto
      });
      
      hlsRef.current = hls;
      hls.loadSource(channel.streamUrl);
      hls.attachMedia(video);
      
      hls.on(Hls.Events.MANIFEST_PARSED, () => {
        setIsLoading(false);
        video.play().catch(() => {
          // Autoplay might be blocked - user will need to click
        });
      });
      
      hls.on(Hls.Events.ERROR, (_, data) => {
        if (data.fatal) {
          setError('Failed to load stream');
          setIsLoading(false);
        }
      });
    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
      // Native HLS (Safari)
      video.src = channel.streamUrl;
      video.addEventListener('loadedmetadata', () => {
        setIsLoading(false);
        video.play().catch(() => {});
      });
    } else {
      setError('HLS not supported in this browser');
      setIsLoading(false);
    }

    return () => {
      if (hlsRef.current) {
        hlsRef.current.destroy();
        hlsRef.current = null;
      }
      if (video) {
        video.pause();
        video.src = '';
      }
    };
  }, [channel.streamUrl]);

  // Auto-dismiss timer
  useEffect(() => {
    let timer = autoDismissSeconds;
    progressInterval.current = setInterval(() => {
      timer -= 0.1;
      setProgress((timer / autoDismissSeconds) * 100);
      
      if (timer <= 0) {
        clearInterval(progressInterval.current);
        onDismiss();
      }
    }, 100);

    return () => {
      if (progressInterval.current) {
        clearInterval(progressInterval.current);
      }
    };
  }, [autoDismissSeconds, onDismiss]);

  // Play/Pause
  const togglePlay = () => {
    const video = videoRef.current;
    if (!video) return;
    
    if (video.paused) {
      video.play();
      setIsPlaying(true);
    } else {
      video.pause();
      setIsPlaying(false);
    }
  };

  // Toggle mute
  const toggleMute = () => {
    const video = videoRef.current;
    if (!video) return;
    
    video.muted = !video.muted;
    setIsMuted(video.muted);
  };

  // Handle video click
  const handleVideoClick = () => {
    togglePlay();
  };

  // Handle watch now
  const handleWatchNow = () => {
    if (hlsRef.current) {
      hlsRef.current.destroy();
    }
    onWatch(channel.id);
  };

  // Keyboard shortcuts
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        onDismiss();
      }
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        togglePlay();
      }
      if (e.key === 'm') {
        toggleMute();
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);

// app/api/preferences/route.ts

import { NextResponse } from 'next/server';
import { auth } from '@/lib/auth/auth';
import { db } from '@/lib/db';
import { userPreferences } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';

export async function GET() {
  try {
    const session = await auth();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const prefs = await db
      .select()
      .from(userPreferences)
      .where(eq(userPreferences.userId, session.user.id))
      .limit(1);

    return NextResponse.json({
      data: prefs[0] || null,
    });
  } catch (error) {
    console.error('Preferences API error:', error);
    return NextResponse.json(
      { error: 'Failed to fetch preferences' },
      { status: 500 }
    );
  }
}

BEGIN
  NEW.last_updated = NOW();
  RETURN NEW;
END;
